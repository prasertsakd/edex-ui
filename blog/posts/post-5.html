Floating, draggable windows are a staple of sci-fi UIs. eDEX-UI has a complete modal window system: windows can be dragged by their title bar, stacked with z-index management, and closed with a blink animation. Here's how to port that to a web page with vanilla JavaScript — no React, no jQuery, no library.

Core Concept

Dragging works by tracking the delta between consecutive mouse positions and adding that delta to the window's left and top CSS properties. We listen on window (not the element itself) during drag so fast mouse movement doesn't escape the element boundary. 

class BlogWindow {
    constructor({ title, bodyHtml, width, height }) {
        this.id   = crypto.randomUUID();
        this.posX = (window.innerWidth  - parseInt(width))  / 2;
        this.posY = (window.innerHeight - parseInt(height)) / 2;
        this._build(title, bodyHtml, width, height);
        this._initDrag();
        this._animateOpen();
    }

    _el() { return document.getElementById(`bwin_${this.id}`); }
}

Building the DOM

_build(title, bodyHtml, width, height) {
    const el = document.createElement('div');
    el.className = 'blog_window';
    el.id = `bwin_${this.id}`;
    el.style.cssText = `
        width: ${width}; height: ${height};
        left: ${this.posX}px; top: ${this.posY}px;
        position: fixed; z-index: 500;
    `;
    el.innerHTML = `
        <div class="window_titlebar" id="titlebar_${this.id}">
            <h1>${title}</h1>
            <button onclick="this.closest('.blog_window')._winInstance.close()">[ X ]</button>
        </div>
        <div class="window_body">${bodyHtml}</div>
    `;
    el._winInstance = this;
    document.getElementById('modal_layer').appendChild(el);
}

Mouse Drag Handler

_initDrag() {
    const el  = this._el();
    const bar = document.getElementById(`titlebar_${this.id}`);
    let startX, startY, startLeft, startTop;

    const onMove = e => {
        this.posX = startLeft + (e.clientX - startX);
        this.posY = Math.max(0, startTop + (e.clientY - startY));
        el.style.left = `${this.posX}px`;
        el.style.top  = `${this.posY}px`;
    };
    const onUp = () => {
        window.removeEventListener('mousemove', onMove);
        window.removeEventListener('mouseup',   onUp);
    };
    bar.addEventListener('mousedown', e => {
        startX = e.clientX; startY = e.clientY;
        startLeft = this.posX; startTop = this.posY;
        window.addEventListener('mousemove', onMove);
        window.addEventListener('mouseup',   onUp);
    });
}

Open & Close Animations

/* CSS */
.blog_window {
    transform: scale(0.88);
    opacity: 0;
    transition: transform .15s ease-out, opacity .15s ease-out;
}
.blog_window.window_open {
    transform: scale(1);
    opacity: 1;
}
.blog_window.window_closing {
    animation: blink 0.1s linear infinite;
}
// JS - open animation
_animateOpen() {
    requestAnimationFrame(() => {
        requestAnimationFrame(() => this._el().classList.add('window_open'));
    });
}

// JS - close with blink
close() {
    const el = this._el();
    el.classList.add('window_closing');
    setTimeout(() => el.remove(), 120);
}

Z-Index Management

When a window is clicked, it should come to the front. Find the highest current z-index among all open windows and set the focused window one higher: 

focus() {
    const maxZ = Object.values(window._openWindows).reduce((acc, w) => {
        const z = parseInt(document.getElementById(`bwin_${w.id}`)?.style.zIndex || 500);
        return Math.max(acc, z);
    }, 500);
    this._el().style.zIndex = maxZ + 1;
}
