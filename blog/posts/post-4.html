xterm.js is the open-source terminal emulator that powers VS Code's integrated terminal, JupyterLab, and many other browser-based tools. It implements a full VT100/xterm-compatible terminal — ANSI escape codes, color, cursor movement, alternate screen buffer, mouse events — entirely in JavaScript.

Basic Setup

<!-- Load via CDN -->
<link rel="stylesheet" href="https://unpkg.com/xterm@4/css/xterm.css">
<script src="https://unpkg.com/xterm@4/lib/xterm.js"></script>

<div id="terminal"></div>

<script>
const term = new Terminal({
    fontFamily: 'Fira Mono, monospace',
    fontSize: 14,
    cursorStyle: 'block',
    cursorBlink: true,
    theme: {
        background: '#05080d',
        foreground: '#aacfd1',
        cursor:     '#aacfd1',
    }
});
term.open(document.getElementById('terminal'));
term.write('Hello, terminal!\r\n$ ');
</script>

Handling Keyboard Input

term.onData(callback) fires for every key press with the raw terminal sequence. You need to implement your own line editor — buffering characters until Enter is pressed: 

let inputBuf = '';

term.onData(data => {
    const code = data.charCodeAt(0);

    if (code === 13) {              // Enter
        term.write('\r\n');
        processCommand(inputBuf.trim());
        inputBuf = '';

    } else if (code === 127) {      // Backspace
        if (inputBuf.length > 0) {
            inputBuf = inputBuf.slice(0, -1);
            term.write('\b \b');    // erase character
        }

    } else if (code >= 32) {        // Printable characters
        inputBuf += data;
        term.write(data);
    }
});

Command History

Arrow keys send escape sequences (\x1b[A for up, \x1b[B for down). A simple ring buffer gives you bash-style history: 

const history = [];
let histIdx = -1;

// In onData handler:
if (data === '\x1b[A' && histIdx < history.length - 1) {
    histIdx++;
    replaceInput(history[histIdx]);
} else if (data === '\x1b[B' && histIdx > -1) {
    histIdx = Math.max(-1, histIdx - 1);
    replaceInput(histIdx >= 0 ? history[histIdx] : '');
}

function replaceInput(next) {
    term.write('\b \b'.repeat(inputBuf.length));
    term.write(next);
    inputBuf = next;
}

ANSI Escape Codes for Styling

term.writeln('\x1b[1mBold text\x1b[0m');
term.writeln('\x1b[32mGreen text\x1b[0m');
term.writeln('\x1b[31mError message\x1b[0m');
term.writeln('\x1b[33mWarning\x1b[0m');
term.writeln('\x1b[2mDimmed\x1b[0m');
